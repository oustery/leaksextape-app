   458→   458→  /// Parse pagination info with safe defaults
   459→   459→  static Map<String, int> parsePagination(dom.Document document) {
   460→   460→    const defaultResult = {'pages': 1, 'total': 100};
   461→   461→    
   462→   462→    try {
   463→   463→      final paginationEl = document.querySelector(
   464→   464→        '.pagination, .paging, [class*="pagination"], nav'
   465→   465→      );
   466→   466→      
   467→   467→      if (paginationEl == null) return defaultResult;
   468→   468→      
   469→   469→      final pageElements = paginationEl.querySelectorAll('a, li, span, button');
   470→   470→      final pages = pageElements.where((el) => 
   471→   471→        el.text.trim().isNotEmpty && _numericOnly.hasMatch(el.text.trim())
   472→   472→      ).length;
   473→   473→      
   474→   474→      int total = 100;
   475→   475→      final text = paginationEl.text;
   476→   476→      final totalMatches = RegExp(r'(?:total|of|results?)[:\s]*(\d+)', caseSensitive: false)
   477→   477→          .allMatches(text);
   478→   478→      
   479→   479→      if (totalMatches.isNotEmpty) {
   480→   480→        final lastMatch = totalMatches.last.group(1);
   481→   481→        if (lastMatch != null) {
   482→   482→          total = int.tryParse(lastMatch) ?? 100;
   483→   483→        }
   484→   484→      }
   485→   485→      
   486→   486→      if (total <= 0) {
   487→   487→        final numbers = RegExp(r'\b(\d{3,})\b').allMatches(text)
   488→   488→            .map((m) => int.tryParse(m.group(1) ?? '') ?? 0)
   489→   489→            .where((n) => n > total)
   490→   490→            .toList();
   491→   491→        if (numbers.isNotEmpty) {
   492→   492→          total = numbers.last;
