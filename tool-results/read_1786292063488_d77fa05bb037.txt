    90→  /// Parse video item - OPTIMIZED for leak-sex-tape.com structure
    91→  static VideoItem? _parseVideoItem(dom.Element itemElement) {
    92→    try {
    93→      // Find the anchor tag with video link
    94→      final anchor = itemElement.querySelector('a[href*="/video/"]');
    95→      if (anchor == null) return null;
    96→
    97→      final href = anchor.attributes['href'] ?? '';
    98→      if (href.isEmpty) return null;
    99→
   100→      // Extract video ID from URL: /video/1513/slug/
   101→      final id = _extractVideoId(href);
   102→      if (id == null || id.isEmpty) return null;
   103→
   104→      // Extract title from <strong class="title"> or title attribute
   105→      String title = '';
   106→      final titleEl = itemElement.querySelector('strong.title');
   107→      if (titleEl != null) {
   108→        title = titleEl.text.trim();
   109→      }
   110→      if (title.isEmpty) {
   111→        title = anchor.attributes['title'] ?? '';
   112→      }
   113→      if (title.isEmpty) return null;
   114→
   115→      // Extract thumbnail - CRITICAL FIX: use data-original or data-webp
   116→      String thumbnailUrl = '';
   117→      final img = itemElement.querySelector('img.thumb, img[data-original], img[data-webp], img.lazy-load');
   118→      if (img != null) {
   119→        // Priority: data-webp (WebP is smaller) > data-original > src
   120→        thumbnailUrl = img.attributes['data-webp'] ?? 
   121→                       img.attributes['data-original'] ?? 
   122→                       img.attributes['src'] ?? '';
   123→        
   124→        // Skip placeholder images (1x1 GIF)
   125→        if (_isPlaceholderImage(thumbnailUrl)) {
   126→          thumbnailUrl = '';
   127→        }
   128→      }
   129→
   130→      // Extract duration from .duration
   131→      String duration = '';
   132→      final durationEl = itemElement.querySelector('.duration');
   133→      if (durationEl != null) {
   134→        duration = durationEl.text.trim();
   135→      }
   136→
   137→      // Extract views from .views (e.g., "574.8k" -> 574800)
   138→      int views = 0;
   139→      final viewsEl = itemElement.querySelector('.views');
   140→      if (viewsEl != null) {
   141→        views = _parseViewCount(viewsEl.text.trim());
   142→      }
   143→
   144→      // Extract rating from .rating (e.g., "84%" -> 84.0)
   145→      double rating = 0.0;
   146→      final ratingEl = itemElement.querySelector('.rating');
   147→      if (ratingEl != null) {
   148→        rating = _parseRating(ratingEl.text.trim());

   152→        id: id,
   153→        title: _sanitizeString(title, maxLength: _maxTitleLength),
   154→        thumbnailUrl: _normalizeImageUrl(thumbnailUrl),
   155→        duration: duration,
   156→        views: views,
   157→        rating: rating,
   158→        dateAdded: DateTime.now().toIso8601String(),
   159→      );
   160→    } catch (e) {
   161→      debugPrint('HtmlParser: Error in _parseVideoItem: $e');
   162→      return null;
   163→    }

   283→  /// Extract video source from flashvars JavaScript - PRIMARY METHOD
   284→  static VideoSource? _extractSourceFromFlashvars(dom.Document document) {
   285→    final scripts = document.querySelectorAll('script:not([src])');
   286→    
   287→    for (final script in scripts) {
   288→      final content = script.text;
   289→      
   290→      // Check if this script contains video-related data
   291→      if (!content.contains('video_url') && 
   292→          !content.contains('video_url:') &&
   293→          !content.contains('get_file')) {
   294→        continue;
   295→      }
   296→
   297→      // Pattern 1: video_url: 'URL' or video_url: "URL"
   298→      final pattern1 = RegExp("video_url[\\s]*:[\\s]*['\"]([^'\"]+)['\"]", caseSensitive: false);
   299→      final pattern2 = RegExp("video_url[\\s]*:[\\s]*['\"]([^'\"]*get_file[^'\"]*)['\"]", caseSensitive: false);
   300→      final pattern3 = RegExp(r"'(https?://[^']*\.mp4[^']*)'", caseSensitive: false);
   301→      final pattern4 = RegExp(r'"(https?://[^"]*\.mp4[^"]*)"', caseSensitive: false);
   302→      final pattern5 = RegExp("embed['\"\\s/]+(\d+)", caseSensitive: false);
   303→      
   304→      final List<RegExp> patterns = [pattern1, pattern2, pattern3, pattern4, pattern5];
   305→
   306→      for (final pattern in patterns) {
   307→        try {
   308→          final match = pattern.firstMatch(content);
   309→          if (match != null) {
   310→            var videoUrl = match.group(1) ?? '';
   311→            
   312→            if (videoUrl.isEmpty) continue;
   313→            
   314→            // If we got just an ID from embed pattern, construct full URL
   315→            if (RegExp(r'^\d+$').hasMatch(videoUrl)) {
   316→              videoUrl = '${AppConstants.baseUrl}/embed/$videoUrl';
   317→            }
   318→            
   319→            // Process relative URLs
   320→            videoUrl = _normalizeUrl(videoUrl);
   321→            
   322→            // Validate before returning
   323→            if (videoUrl.isNotEmpty && _isValidVideoUrl(videoUrl)) {
   324→              return VideoSource(
   325→                videoUrl: videoUrl,
   326→                quality: _detectQuality(content),
   327→                format: _detectFormat(videoUrl),
   328→              );
   329→            }
   330→          }
   331→        } catch (e) {
   332→          continue;
   333→        }
   334→      }
   335→    }
   336→    
   337→    return null;
   338→  }

   458→  /// Parse pagination info with safe defaults
   459→  static Map<String, int> parsePagination(dom.Document document) {
   460→    const defaultResult = {'pages': 1, 'total': 100};
   461→    
   462→    try {
   463→      final paginationEl = document.querySelector(
   464→        '.pagination, .paging, [class*="pagination"], nav'
   465→      );
   466→      
   467→      if (paginationEl == null) return defaultResult;
   468→      
   469→      final pageElements = paginationEl.querySelectorAll('a, li, span, button');
   470→      final pages = pageElements.where((el) => 
   471→        el.text.trim().isNotEmpty && _numericOnly.hasMatch(el.text.trim())
   472→      ).length;
   473→      
   474→      int total = 100;
   475→      final text = paginationEl.text;
   476→      final totalMatches = RegExp(r'(?:total|of|results?)[:\s]*(\d+)', caseSensitive: false)
   477→          .allMatches(text);
   478→      
   479→      if (totalMatches.isNotEmpty) {
   480→        final lastMatch = totalMatches.last.group(1);
   481→        if (lastMatch != null) {
   482→          total = int.tryParse(lastMatch) ?? 100;
   483→        }
   484→      }
   485→      
   486→      if (total <= 0) {
   487→        final numbers = RegExp(r'\b(\d{3,})\b').allMatches(text)
   488→            .map((m) => int.tryParse(m.group(1) ?? '') ?? 0)
   489→            .where((n) => n > total)
   490→            .toList();
   491→        if (numbers.isNotEmpty) {
   492→          total = numbers.last;
   493→        }
   494→      }
   495→      
   496→      return {
   497→        'pages': pages.clamp(1, 9999),
   498→        'total': total.clamp(1, 99999),
   499→      };
   500→    } catch (e) {
   501→      debugPrint('HtmlParser: Error parsing pagination: $e');
   502→      return defaultResult;
   503→    }
   504→  }

   516→      RegExp(r'[?&](?:v|id|video_id)=(\d+)'),
   517→      RegExp(r'/(\d{4,})(?:\.html?|/|$)'),
   518→    ];
   519→    
   520→    for (final pattern in patterns) {
   521→      final match = pattern.firstMatch(href);
   522→      if (match != null) {
   523→        return match.group(1);
   524→      }
   525→    }
   526→    
   527→    return null;
   528→  }

   612→  static int _extractViews(dom.Element element) {
   613→    final selectors = [
   614→      '.views', '.view-count', '.hits',
   615→      '[class*="views"]', '[class*="hits"]',
   616→      '[data-views]',
   617→    ];
   618→    
   619→    for (final selector in selectors) {
   620→      final el = element.querySelector(selector);
   621→      if (el != null) {
   622→        return _parseViewCount(el.text.trim());
   623→      }
   624→    }
   625→    
   626→    return 0;
   627→  }

   629→  /// Parse view count string to integer (handles "574.8k", "1.2M", etc.)
   630→  static int _parseViewCount(String text) {
   631→    if (text.isEmpty) return 0;
   632→    
   633→    final cleaned = text.replaceAll(',', '').replaceAll('.', '').toLowerCase();
   634→    final numericPart = cleaned.replaceAll(RegExp(r'[^\d]'), '');
   635→    final views = int.tryParse(numericPart) ?? 0;
   636→    
   637→    // Handle multipliers
   638→    if (text.contains('k')) {
   639→      return (views * 1000).round();
   640→    } else if (text.contains('m')) {
   641→      return (views * 1000000).round();
   642→    }
   643→    
   644→    return views;
   645→  }

   665→  /// Parse rating string to double (handles "84%", "4.5", etc.)
   666→  static double _parseRating(String text) {
   667→    if (text.isEmpty) return 0.0;
   668→    
   669→    var cleaned = text.replaceAll('%', '').trim();
   670→    cleaned = cleaned.replaceAll(RegExp(r'[^\d.]'), '');
   671→    final rating = double.tryParse(cleaned);
   672→    
   673→    if (rating != null && rating >= 0) {
   674→      // Normalize to 0-100 scale if needed
   675→      return rating <= 5 ? rating * 20 : rating;
   676→    }
   677→    
   678→    return 0.0;
   679→  }

   701→  /// Detect video format from URL
   702→  static String _detectFormat(String url) {
   703→    if (url.contains('.m3u8')) return 'hls';
   704→    if (url.contains('.webm')) return 'webm';

   734→  /// Check if URL is valid for video playback
   735→  static bool _isValidVideoUrl(String url) {
   736→    if (url.isEmpty) return false;
   737→    
   738→    final uri = Uri.tryParse(url);
   739→    if (uri == null) return false;
   740→    
   741→    if (uri.hasScheme && _invalidSchemes.contains(uri.scheme.toLowerCase())) {
   742→      return false;
   743→    }
   744→    
   745→    if (uri.hasScheme && !['http', 'https'].contains(uri.scheme.toLowerCase())) {
   746→      return false;
   747→    }
   748→    
   749→    if (_validVideoExtensions.hasMatch(url)) {
   750→      return true;
   751→    }
   752→    
   753→    if (url.contains('embed') || url.contains('player') || url.contains('stream')) {
   754→      return true;
   755→    }
   756→    
   757→    if (url.contains('get_file')) {  // leak-sex-tape.com specific
   758→      return true;
   759→    }
   760→    
   761→    if (uri.host.isNotEmpty && uri.host.contains('.')) {
   762→      return true;
   763→    }
   764→    
   765→    return false;
   766→  }

   768→  /// General URL validation
   769→  static bool _isValidUrl(String url) {
   770→    if (url.isEmpty) return false;
   771→    
   772→    final uri = Uri.tryParse(url);
   773→    if (uri == null) return false;
   774→    
   775→    if (uri.hasScheme && _invalidSchemes.contains(uri.scheme.toLowerCase())) {
   776→      return false;
   777→    }
   778→    
   779→    return uri.host.isNotEmpty || url.startsWith('/');
   780→  }

   787→      'empty',

   799→  /// Sanitize string input
   800→  static String _sanitizeString(String input, {int maxLength = 100}) {
   801→    if (input.isEmpty) return input;
   802→    
   803→    var sanitized = input.trim();
   804→    
   805→    if (sanitized.length > maxLength) {
   806→      sanitized = sanitized.substring(0, maxLength);
   807→    }
   808→    
   809→    sanitized = sanitized.replaceAll(RegExp(r'[\x00-\x08\x0B\x0C\x0E-\x1F]'), '');
   810→    
   811→    return sanitized;
   812→  }

   814→  /// Extract category ID from href
   815→  static String _extractCategoryId(String href, int fallbackIndex) {
   816→    if (href.isEmpty) return fallbackIndex.toString();
   817→    
   818→    final segments = href.split('/').where((s) => s.isNotEmpty).toList();
   819→    if (segments.isNotEmpty) {
   820→      final lastSegment = segments.last;
   821→      return lastSegment.replaceAll(RegExp(r'\.html?$'), '');
   822→    }
   823→    
   824→    return fallbackIndex.toString();
   825→  }
